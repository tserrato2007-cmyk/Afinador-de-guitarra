#include <Arduino.h>
#include <U8g2lib.h>
#include <Wire.h>

// Configuración de la pantalla OLED SH1106 de 1.3"
U8G2_SH1106_128X64_NONAME_F_HW_I2C u8g2(U8G2_R0, /* reset=*/ U8X8_PIN_NONE);

const int micPin = 36; // Pin VP del ESP32

struct StringNote {
  const char* name;
  float freq;
  float tolerance; // Tolerancia específica en Hz para cada cuerda
};

StringNote openStrings[6] = {
  {"6E",  82.41, 2.0},
  {"5A", 110.00, 2.0},
  {"4D", 146.83, 2.2},
  {"3G", 196.00, 2.5},
  {"2B", 246.94, 3.0}, // Mayor tolerancia para agudos por la física de la cuerda
  {"1E", 329.63, 3.5}
};

const int numSamples = 256;
int rawAudio[numSamples];
float correlationBuffer[numSamples / 2];
int dcOffset = 0;

void setup() {
  Serial.begin(115200);
  delay(500);
  
  Wire.begin(21, 22);
  u8g2.begin();
  
  analogSetAttenuation(ADC_11db);
  pinMode(micPin, INPUT);
}

float getGuitarFrequency() {
  unsigned long startTime = micros();
  
  // Muestreo a máxima velocidad sin delay fijo para mayor resolución en agudos
  for (int i = 0; i < numSamples; i++) {
    rawAudio[i] = analogRead(micPin);
    delayMicroseconds(100); // 100us da ~7.5kHz de muestreo, excelente para 1E y 2B
    if (i % 64 == 0) yield();
  }
  
  unsigned long duration = micros() - startTime;
  if (duration == 0) return 0;

  float samplingFreq = (numSamples * 1000000.0) / duration;

  // 1. DC Offset
  long sum = 0;
  for (int i = 0; i < numSamples; i++) sum += rawAudio[i];
  dcOffset = sum / numSamples;

  // 2. Amplitud pico a pico
  int minVal = 4095, maxVal = 0;
  for (int i = 0; i < numSamples; i++) {
    if (rawAudio[i] < minVal) minVal = rawAudio[i];
    if (rawAudio[i] > maxVal) maxVal = rawAudio[i];
  }
  int amplitude = maxVal - minVal;

  if (amplitude < 20) return 0; // Umbral de silencio

  // 3. Autocorrelación en el rango 70 Hz - 380 Hz
  int minLag = (int)(samplingFreq / 380.0);
  int maxLag = (int)(samplingFreq / 70.0);
  
  if (minLag < 2) minLag = 2;
  if (maxLag >= numSamples / 2) maxLag = numSamples / 2 - 1;

  float bestCorrelation = -1;
  int bestLag = 0;

  // Limpiar buffer de correlación
  for (int i = 0; i < numSamples / 2; i++) correlationBuffer[i] = 0;

  for (int lag = minLag; lag <= maxLag; lag++) {
    float correlation = 0;
    for (int i = 0; i < (numSamples - lag); i++) {
      correlation += (float)(rawAudio[i] - dcOffset) * (float)(rawAudio[i + lag] - dcOffset);
    }
    correlationBuffer[lag] = correlation;

    if (correlation > bestCorrelation) {
      bestCorrelation = correlation;
      bestLag = lag;
    }
  }

  if (bestLag <= minLag || bestLag >= maxLag) return 0;

  // 4. INTERPOLACIÓN PARABÓLICA (Ajuste fino para eliminar saltos discretos)
  float alpha = correlationBuffer[bestLag - 1];
  float beta  = correlationBuffer[bestLag];
  float gamma = correlationBuffer[bestLag + 1];

  float delta = (gamma - alpha) / (2.0 * (2.0 * beta - alpha - gamma));
  float exactLag = bestLag + delta;

  float frequency = samplingFreq / exactLag;

  if (frequency < 65.0 || frequency > 380.0) return 0;

  return frequency;
}

void loop() {
  float freq = getGuitarFrequency();

  const char* closestNote = "--";
  float targetFreq = 0;
  float currentTolerance = 2.0;
  int status = 0; // -1: SUBIR, 1: BAJAR, 0: [AFINADA]
  bool valid = false;

  if (freq > 0) {
    float minDiff = 999.0;
    int bestIndex = 0;

    for (int i = 0; i < 6; i++) {
      float diff = abs(freq - openStrings[i].freq);
      if (diff < minDiff) {
        minDiff = diff;
        bestIndex = i;
      }
    }

    // Aceptamos la cuerda si está dentro de un rango de búsqueda amplio (ej. 30 Hz)
    if (minDiff <= 30.0) {
      closestNote = openStrings[bestIndex].name;
      targetFreq = openStrings[bestIndex].freq;
      currentTolerance = openStrings[bestIndex].tolerance;
      valid = true;

      // Evaluación de precisión usando la interpolación
      if (freq < (targetFreq - currentTolerance)) {
        status = -1; // SUBIR
      } else if (freq > (targetFreq + currentTolerance)) {
        status = 1;  // BAJAR
      } else {
        status = 0;  // [AFINADA]
      }
    }
  }

  // Refrescar pantalla OLED SH1106
  u8g2.clearBuffer();
  
  // 1. Dibujar onda estacionaria en la mitad inferior (Y: 45 a 63)
  u8g2.drawHLine(0, 54, 128); 
  for (int i = 0; i < 127; i++) {
    int idx1 = map(i, 0, 127, 0, numSamples - 1);
    int idx2 = map(i + 1, 0, 127, 0, numSamples - 1);

    int y1 = map(rawAudio[idx1] - dcOffset, -2048, 2048, 63, 45);
    int y2 = map(rawAudio[idx2] - dcOffset, -2048, 2048, 63, 45);

    u8g2.drawLine(i, y1, i + 1, y2);
  }

  // 2. Interfaz superior
  if (valid) {
    u8g2.setFont(u8g2_font_fub25_tr);
    u8g2.setCursor(5, 36);
    u8g2.print(closestNote);

    u8g2.setFont(u8g2_font_ncenB08_tr);
    char hzStr[16];
    sprintf(hzStr, "%.1f Hz", freq);
    u8g2.setCursor(65, 15);
    u8g2.print(hzStr);

    u8g2.setCursor(55, 33);
    if (status == -1) {
      u8g2.print("<< SUBIR");
    } else if (status == 1) {
      u8g2.print("BAJAR >>");
    } else {
      u8g2.print("[AFINADA]");
    }
  } else {
    u8g2.setFont(u8g2_font_ncenB08_tr);
    u8g2.setCursor(15, 25);
    u8g2.print("Toca una cuerda");
  }

  u8g2.sendBuffer();
  delay(15);
}
